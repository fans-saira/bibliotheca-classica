\newif\ifprimeiroverso
\def\verso#1#2#3{%
  \ifprimeiroverso\nointerlineskip\fi
  \vbox{\hsize=\textwidth\hangindent=6mm\hangafter=1
    \ifprimeiroverso\setupinterlinespace[line=18pt]\fi
    \noindent\setmarking[verse]{#1}%
    \doifnotempty{#2}{\medirnota{#1}{#2}\inouter{#2}}%
    \doifelse{#1}{1}{\numeroverso{#1}}%
      {\ifnum\numexpr#1-5*(#1/5)\relax=0 \numeroverso{#1}\else\numeroverso{}\fi}%
    \hbox{#3}\par}\global\primeiroversofalse}
